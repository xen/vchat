import hashlib
import hmac
import os
import secrets
import time

import requests

api_url = "https://local.vchat.com/api/update"
client_id = os.environ["VCHAT_CLIENT_ID"]
client_secret = os.environ["VCHAT_CLIENT_SECRET"]
url = "https://example.com/docs/page"

timestamp = str(int(time.time()))
nonce = secrets.token_hex(16)
form = {
    "url": url,
    "client_id": client_id,
    "timestamp": timestamp,
    "nonce": nonce,
}
payload = "\n".join(f"{key}={value}" for key, value in sorted(form.items()))
signature = hmac.new(
    client_secret.encode("utf-8"),
    payload.encode("utf-8"),
    hashlib.sha256,
).hexdigest()
form["signature"] = signature

response = requests.post(
    api_url,
    data=form,
    timeout=30,
)
response.raise_for_status()
print(response.json())
PHP
<?php
$apiUrl = 'https://local.vchat.com/api/update';
$clientId = getenv('VCHAT_CLIENT_ID');
$clientSecret = getenv('VCHAT_CLIENT_SECRET');
$url = 'https://example.com/docs/page';

if (!$clientId || !$clientSecret) {
    throw new RuntimeException('Set VCHAT_CLIENT_ID and VCHAT_CLIENT_SECRET');
}

$timestamp = (string) time();
$nonce = bin2hex(random_bytes(16));
$form = [
    'url' => $url,
    'client_id' => $clientId,
    'timestamp' => $timestamp,
    'nonce' => $nonce,
];
ksort($form);
$payload = implode("\n", array_map(
    static fn ($key, $value) => "{$key}={$value}",
    array_keys($form),
    $form
));
$signature = hash_hmac('sha256', $payload, $clientSecret);
$form['signature'] = $signature;

$body = http_build_query($form, '', '&');

$ch = curl_init($apiUrl);
curl_setopt_array($ch, [
    CURLOPT_POST => true,
    CURLOPT_POSTFIELDS => $body,
    CURLOPT_HTTPHEADER => ['Content-Type: application/x-www-form-urlencoded'],
    CURLOPT_RETURNTRANSFER => true,
    CURLOPT_TIMEOUT => 30,
]);

$response = curl_exec($ch);
if ($response === false) {
    throw new RuntimeException(curl_error($ch));
}
$status = curl_getinfo($ch, CURLINFO_RESPONSE_CODE);
curl_close($ch);

if ($status < 200 || $status >= 300) {
    throw new RuntimeException($response);
}

echo $response;
